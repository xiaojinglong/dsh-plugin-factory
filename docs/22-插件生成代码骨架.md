## 二、`service/api-runner.js`（通用执行层，一次写好后各插件共用）

```js
'use strict';
const { withRetry } = require('./util');
const DEFAULTS = { baseUrl: 'https://api.giikin.com', authEnv: 'LISTING_API_KEY', timeoutMs: 60000, maxPages: 10, pageSize: 20 };

async function postJson(url, authKey, body, timeoutMs) {
  const res = await fetch(url, { method: 'POST', headers: { 'content-type': 'application/json', authorization: `Bearer ${authKey}` }, body: JSON.stringify(body), signal: AbortSignal.timeout(timeoutMs) });
  const text = await res.text();
  let j; try { j = JSON.parse(text); } catch { throw new Error(`接口返回非 JSON（HTTP ${res.status}）：${text.slice(0, 120)}`); }
  if (j.code !== 200) throw new Error(`接口业务失败 code=${j.code}：${String(j.message || j.msg || '').slice(0, 160)}`);
  return j.data || {};
}

async function runTask(job, ctx, cfgIn) {
  const cfg = { ...DEFAULTS, ...cfgIn };
  const p = ctx.params || {};
  const authKey = String(p.auth_key || process.env[cfg.authEnv] || '').trim();
  if (!authKey) throw new Error(`缺少凭证：params.auth_key 或环境变量 ${cfg.authEnv}`);
  const pages = Math.max(1, Math.min(cfg.maxPages, Number(p.pages) || 2));
  const pageSize = Math.max(1, Math.min(200, Number(p.pageSize) || cfg.pageSize));
  const url = `${cfg.baseUrl}${cfg.path}`;
  const pagesData = []; let total = 0, totalPages = 0, extra = {};
  for (let page = 1; page <= pages; page++) {
    if (ctx.isAborted()) throw new Error('任务已被终止');
    ctx.setProgress(`拉取第 ${page}/${pages} 页（${cfg.task}）`);
    const data = await withRetry(() => postJson(url, authKey, cfg.buildBody(p, page, pageSize), cfg.timeoutMs),
      { retries: 1, delayMs: 900, label: `第 ${page} 页` });
    const pag = data.pagination || {};
    total = Number(pag.total ?? data.total ?? total); totalPages = Number(pag.totalPages ?? totalPages);
    const rows = Array.isArray(data.list) ? data.list : (Array.isArray(data.campaigns) ? data.campaigns : []);
    extra = { ...extra, ...Object.fromEntries(Object.entries(data).filter(([k]) => !['list', 'campaigns', 'pagination'].includes(k))) };
    if (!rows.length) { ctx.log(`第 ${page} 页无数据`); break; }
    pagesData.push({ page, rows });
  }
  const dataset = { task: cfg.task, fetchedAt: new Date().toISOString(), params: { ...p, pages: pagesData.length, pageSize }, extra, total, totalPages, pages: pagesData };
  ctx.setProgress('整理产物');
  await ctx.saveFile({ kind: 'main', fileName: `${cfg.task}_dataset.json`, content: JSON.stringify(dataset, null, 2), encoding: 'utf8' });
  ctx.log(`完成：${pagesData.length} 页 / ${pagesData.reduce((s, x) => s + x.rows.length, 0)} 条 / 全量 ${total}`);
}
module.exports = { runTask };
```

## 三、`service/fetcher.js`（每插件只写这一段）

```js
'use strict';
const { runTask } = require('./api-runner');
const TASK = {
  task: '<TASK>',
  path: '/agent/shopee/user/openapi/<endpoint>',
  authEnv: 'LISTING_API_KEY',
  pageSize: 20,        // 按接口上限
  maxPages: 10,
  buildBody: (p, page, pageSize) => ({
    page, pageSize,
    // ……业务参数，含默认值（如 shop_user_name: String(p.shop_user_name || '陈雄德')）
  }),
};
async function runJob(job, ctx) { return runTask(job, ctx, TASK); }
module.exports = { runJob };
```

`server.js` 只需两处改动（其余从 `D:\www\dshspirder\src` 复制）：
`const crawler = require('./crawler')` → `require('./fetcher')`；`manager.setRunner(crawler.runJob)` → `fetcher.runJob`；并删掉浏览器预热/关闭两行。用 `patch-server.mjs` 自动改：

```js
// service/patch-server.mjs
const fs = require('fs');
let s = fs.readFileSync('server.js', 'utf8');
s = s.replace("const crawler = require('./crawler');", "const fetcher = require('./fetcher');");
s = s.replace('manager.setRunner(crawler.runJob)', 'manager.setRunner(fetcher.runJob)');
s = s.replace(/\/\/ 预热浏览器[\s\S]*?crawler\.getBrowser\(\)[\s\S]*?\n/, '');
s = s.replace('await crawler.closeBrowser();', 'await Promise.resolve();');
fs.writeFileSync('server.js', s, 'utf8');
// 断言：替换必须全部生效，否则非 0 退出（避免"看起来改了"）
const bad = s.includes("require('./crawler')") || s.includes('crawler.runJob') || s.includes('crawler.getBrowser');
console.log('patch-server:', bad ? 'FAILED' : 'ok');
process.exit(bad ? 1 : 0);
```

## 五、报告生成器 `gen-report.mjs`（核心骨架）

```js
import { writeFile } from 'node:fs/promises';
import path from 'node:path';
import { shell, table, distTable, bar, top, pct } from '../factory-kit/report-kit.mjs';
import { resolveDataset, rowsOf } from '../factory-kit/dataset.mjs';

const PLUGIN_DIR = 'D:/www/dsh-plugin-factory/<id>';
const OUT = path.join(PLUGIN_DIR, 'report', '<id>-report.html');
const { ds, jobId = '', source } = await resolveDataset(PLUGIN_DIR);
const rows = rowsOf(ds);
if (!rows.length) { console.error('[SKIP] 数据集为空（0 条）：不生成报告，保留旧报告。'); process.exit(2); }
const total = Array.isArray(ds) ? (ds.find?.((p) => p.total)?.total ?? 0) : (ds.total ?? 0);
const fetchedAt = ds.fetchedAt || '';
const jobLabel = jobId || path.basename(path.dirname(source || ''));

// ---- 统计（按业务改）----
const dist = top(rows, (r) => r.site);
const risk = rows.filter((r) => /* 业务风险规则 */ false);

const body = `
<section><h2>一、结论摘要</h2><ul><li>…</li></ul></section>
<section><h2>二、数据口径</h2><table><tbody>
  <tr><th>接口</th><td><code>POST …</code></td></tr>
  <tr><th>参数</th><td>…</td></tr>
  <tr><th>公式</th><td>…（必须写明唯一口径）</td></tr>
  <tr><th>限制</th><td>样本覆盖率 ${rows.length}/${total}；…</td></tr>
</tbody></table></section>
<section><h2>三、结构分布</h2><div class="grid2">
  <div><h3>站点</h3>${distTable(['站点','数量','占比','分布'], dist, rows.length)}</div>
  <div><h3>（其它维度）</h3>…</div></div></section>
<section><h2>四、风险与异常</h2>${table(['级别','类型','对象','说明'], risk.map((x) => `<tr><td>…</td></tr>`))}</section>
<section><h2>五、优化建议</h2><ul><li>…（每条带数据依据）</li></ul></section>
<section><h2>六、限制与未决</h2><ul><li>样本仅 ${rows.length} 条（覆盖率 ${pct(rows.length, total)}%）</li><li>…</li></ul></section>
<section><h2>七、附录</h2><table><tbody>
  <tr><th>数据来源</th><td><code>${source}</code></td></tr>
  <tr><th>任务</th><td><code>${jobLabel}</code> 取数 ${fetchedAt || '未记录'}</td></tr>
</tbody></table></section>
<section><h2>八、生成信息</h2><p>插件 <code><id></code> · 数据源 api.giikin.com · 单文件报告</p></section>`;

await writeFile(OUT, shell({
  title: '<中文名>报告',
  meta: ['业务：<id>', '数据源：…', `任务：${jobLabel}`, `取数时间：${fetchedAt || '未记录'}`],
  kpis: [ { k: '样本', v: rows.length, s: `全量 ${total}` }, /* … */ ],
  body,
  footer: `<id> 插件 · ${jobLabel}`,
}), 'utf8');
console.log('报告已生成:', OUT, '| rows:', rows.length, '| total:', total);
```

要求：空数据守卫、页眉 jobId/取数时间、8 节、公式唯一、限制具体。

## 九、常见坑（生成后自查）

| 坑 | 检查 |
|---|---|
| 报告读旧数据 | 页眉必须有「任务/取数时间」；脚本支持 `[dataset.json]` 入参 |
| 库存等口径矛盾 | 报告 §二写唯一公式，§一/§四/§六 同源 |
| 参数被静默忽略 | 写进 SKILL 的「被忽略参数」 |
| `&&` 在 PS 5.1 报错 | 用 `;` |
| `.ps1` 乱码 | UTF-8 带 BOM |
| MCP 改错位置 | 改 `index.js` 的 `SERVER_NAME` + `config.json` 的 baseUrl，两处都要 |
| 端口/命名冲突 | 先查 registry；serverName/taskType 全局唯一 |
| 空数据覆盖报告 | 生成器 `rows.length===0 → exit 2` |
